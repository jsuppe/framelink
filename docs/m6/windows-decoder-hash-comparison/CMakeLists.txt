cmake_minimum_required(VERSION 3.20)
project(FrameLinkHashComparison LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Windows-only standalone feasibility harness")
endif()
set(GSTREAMER_SDK "C:/Program Files/gstreamer/1.0/msvc_x86_64" CACHE PATH "Existing MSVC GStreamer SDK")
find_path(GLIB_INCLUDE glib.h PATHS "${GSTREAMER_SDK}/include/glib-2.0" NO_DEFAULT_PATH REQUIRED)
find_path(GLIB_CONFIG_INCLUDE glibconfig.h PATHS "${GSTREAMER_SDK}/lib/glib-2.0/include" NO_DEFAULT_PATH REQUIRED)
find_library(GLIB_LIBRARY NAMES glib-2.0 PATHS "${GSTREAMER_SDK}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(GLib::GLib UNKNOWN IMPORTED)
set_target_properties(GLib::GLib PROPERTIES IMPORTED_LOCATION "${GLIB_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${GLIB_INCLUDE};${GLIB_CONFIG_INCLUDE}")
add_executable(fl_hash_compare hash_compare.cpp)
target_compile_features(fl_hash_compare PRIVATE cxx_std_17)
target_link_libraries(fl_hash_compare PRIVATE GLib::GLib bcrypt)
if(MSVC)
 target_compile_options(fl_hash_compare PRIVATE /W4 /EHsc)
endif()
